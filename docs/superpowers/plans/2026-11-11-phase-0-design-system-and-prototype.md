# Phase 0: Design System and Clickable Prototype Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Scaffold the `fynix-leads` app, build the portal's own design system with a review page, and build a clickable prototype of three screens (client review queue, lead record, agency lead table) on sample data, so Fynix can approve the look and flow before any backend exists.

**Architecture:** One Next.js 16 App Router app. Domain rules that later phases reuse (status vocabulary, the delivered rule, time zone display, waiting age) are plain TypeScript modules under `lib/` with Vitest tests. UI is shadcn/ui on Base UI, restyled through one token file. The prototype reads typed sample data from `lib/sample/` and keeps state in the browser only. No database, auth or email in this phase.

**Tech Stack:** Next.js 16.4, React 19.3, TypeScript strict, pnpm 11, Tailwind CSS 4.3, shadcn 4.23 (`--base base`), `@base-ui/react` 1.9, `@tanstack/react-table` 9.3, `@phosphor-icons/react` 2.1, `geist` 1.9, `sonner` 2, `cmdk` 1.1, Vitest 5.

**Spec:** `docs/superpowers/specs/2026-11-11-leadgen-portal-design.md`, sections 4, 7, 9, 11 and 13.

**How UI tasks are written.** Logic tasks carry complete code and tests. Screen tasks (9 to 12) carry the exact file list, props, layout rules and an acceptance checklist, plus code for the parts that must be exact (keyboard handling, data wiring). The final markup of a screen is produced by the impeccable shape, build and critique loop against screenshots, because visual work cannot be settled in a document. Every screen task ends with the same QA gate (Task 13).

```ts
const LOCALES_FOR_NAMES = ["en-US", "en-IN", "en-GB"] as const;

function part(date: Date, timeZone: string, locale: string, options: Intl.DateTimeFormatOptions, type: string): string {
  const found = new Intl.DateTimeFormat(locale, { timeZone, ...options })
    .formatToParts(date)
    .find((p) => p.type === type);
  if (!found) throw new Error(`Intl returned no "${type}" part for ${timeZone}`);
  return found.value;
}

/** A readable zone name: "IST", "EST", "BST", else a GMT offset such as "GMT+4". */
export function zoneAbbreviation(date: Date, timeZone: string): string {
  const names = LOCALES_FOR_NAMES.map((l) => part(date, timeZone, l, { timeZoneName: "short" }, "timeZoneName"));
  return names.find((n) => n === "GMT" || !n.startsWith("GMT")) ?? names[0];
}

/** "Thu 12 Nov, 3:00 pm IST" in the given IANA time zone. */
export function formatInZone(date: Date, timeZone: string): string {
  const parts = Object.fromEntries(
    new Intl.DateTimeFormat("en-US", {
      timeZone,
      weekday: "short",
      day: "numeric",
      month: "short",
      hour: "numeric",
      minute: "2-digit",
      hour12: true,
    })
      .formatToParts(date)
      .map((p) => [p.type, p.value]),
  );
  return `${parts.weekday} ${parts.day} ${parts.month}, ${parts.hour}:${parts.minute} ${parts.dayPeriod.toLowerCase()} ${zoneAbbreviation(date, timeZone)}`;
}

/** Spec section 9: warn when a time falls before 8 am or from 7 pm in that zone. */
export function isOutsideWorkingHours(date: Date, timeZone: string, startHour = 8, endHour = 19): boolean {
  const hour = Number(part(date, timeZone, "en-US", { hour: "numeric", hourCycle: "h23" }, "hour"));
  return hour < startHour || hour >= endHour;
}
```

```ts
const MIN = 60_000;
const HOUR = 60 * MIN;
const DAY = 24 * HOUR;

/** Compact waiting time for queue rows: "12m", "5h", "2d 4h", "9d". */
export function formatWaiting(since: Date, now: Date): string {
  const ms = Math.max(0, now.getTime() - since.getTime());
  if (ms < HOUR) return `${Math.floor(ms / MIN)}m`;
  if (ms < DAY) return `${Math.floor(ms / HOUR)}h`;
  const days = Math.floor(ms / DAY);
  if (days >= 7) return `${days}d`;
  const hours = Math.floor((ms % DAY) / HOUR);
  return hours === 0 ? `${days}d` : `${days}d ${hours}h`;
}

/** Spec section 10: the client reminder fires for leads pending over 24 hours. */
export function isOverdue(since: Date, now: Date, hours = 24): boolean {
  return now.getTime() - since.getTime() > hours * HOUR;
}
```

```ts
"use client";
import { useEffect } from "react";

type Handlers = { approve: () => void; reject: () => void; clarify: () => void; next: () => void; previous: () => void };

const KEYS: Record<string, keyof Handlers> = { a: "approve", r: "reject", c: "clarify", j: "next", k: "previous" };

/** Single-key triage shortcuts. Ignored while typing or while a dialog is open. */
export function useReviewKeys(handlers: Handlers, enabled: boolean): void {
  useEffect(() => {
    if (!enabled) return;
    function onKey(e: KeyboardEvent) {
      if (e.metaKey || e.ctrlKey || e.altKey) return;
      const target = e.target as HTMLElement | null;
      if (target?.closest("input, textarea, select, [contenteditable=true], [role=dialog]")) return;
      const action = KEYS[e.key.toLowerCase()];
      if (!action) return;
      e.preventDefault();
      handlers[action]();
    }
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [handlers, enabled]);
}
```

```tsx
"use client";
import { createColumnHelper, tableFeatures, useTable } from "@tanstack/react-table";
import { CaretDown, CaretUp } from "@phosphor-icons/react";
import { useMemo, useState } from "react";
import { StatusBadge } from "@/components/status/status-badge";
import { WaitingAge } from "@/components/time/waiting-age";
import type { Lead } from "@/lib/domain/types";

const features = tableFeatures({});
const helper = createColumnHelper<typeof features, Lead>();

type SortKey = "submittedAt" | "company" | "approval";
type Sort = { key: SortKey; dir: "asc" | "desc" };

const COMPARE: Record<SortKey, (a: Lead, b: Lead) => number> = {
  submittedAt: (a, b) => a.submittedAt.localeCompare(b.submittedAt),
  company: (a, b) => a.prospect.company.localeCompare(b.prospect.company),
  approval: (a, b) => a.approval.localeCompare(b.approval),
};

export function LeadsDataTable({ leads, now }: { leads: Lead[]; now: string }) {
  const [sort, setSort] = useState<Sort>({ key: "submittedAt", dir: "desc" });
  const nowDate = useMemo(() => new Date(now), [now]);

  const data = useMemo(() => {
    const sorted = [...leads].sort(COMPARE[sort.key]);
    return sort.dir === "asc" ? sorted : sorted.reverse();
  }, [leads, sort]);

  const columns = useMemo(
    () =>
      helper.columns([
        helper.accessor("ref", { header: "Ref", cell: (c) => <span className="font-mono">{c.getValue()}</span> }),
        helper.accessor((l) => l.prospect.name, { id: "prospect", header: "Prospect" }),
        helper.accessor((l) => l.prospect.company, { id: "company", header: "Company" }),
        helper.accessor((l) => l.prospect.title, { id: "title", header: "Title" }),
        helper.accessor("approval", {
          header: "Approval",
          cell: (c) => <StatusBadge kind="approval" status={c.getValue()} />,
        }),
        helper.accessor((l) => l.meetings.at(-1)?.status, {
          id: "meeting",
          header: "Meeting",
          cell: (c) => {
            const s = c.getValue();
            return s ? <StatusBadge kind="meeting" status={s} /> : <span className="text-muted-foreground">None</span>;
          },
        }),
        helper.accessor("submittedAt", {
          header: "Submitted",
          cell: (c) => <WaitingAge since={new Date(c.getValue())} now={nowDate} />,
        }),
      ]),
    [nowDate],
  );

  const table = useTable({ features, columns, data });
  const sortable: Partial<Record<string, SortKey>> = { company: "company", approval: "approval", submittedAt: "submittedAt" };

  return (
    <div className="overflow-x-auto rounded-lg border bg-card">
      <table className="w-full text-table">
        <thead className="border-b bg-muted text-left text-muted-foreground">
          {table.getHeaderGroups().map((group) => (
            <tr key={group.id}>
              {group.headers.map((header) => {
                const key = sortable[header.column.id];
                const active = key && sort.key === key;
                return (
                  <th key={header.id} scope="col" className="h-9 px-3 font-medium" aria-sort={active ? (sort.dir === "asc" ? "ascending" : "descending") : undefined}>
                    {key ? (
                      <button
                        type="button"
                        className="inline-flex items-center gap-1 hover:text-foreground"
                        onClick={() => setSort({ key, dir: active && sort.dir === "desc" ? "asc" : "desc" })}
                      >
                        <table.FlexRender header={header} />
                        {active && (sort.dir === "asc" ? <CaretUp size={12} /> : <CaretDown size={12} />)}
                      </button>
                    ) : (
                      <table.FlexRender header={header} />
                    )}
                  </th>
                );
              })}
            </tr>
          ))}
        </thead>
        <tbody className="divide-y">
          {table.getRowModel().rows.map((row) => (
            <tr key={row.id} className="h-10 hover:bg-[var(--gray-2)]">
              {row.getAllCells().map((cell) => (
                <td key={cell.id} className="px-3 whitespace-nowrap">
                  <table.FlexRender cell={cell} />
                </td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}
```
